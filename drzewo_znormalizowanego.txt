│
├── schema_version: string
├── source_file: string
├── group_id: string
├── dataset: string
├── instrument: string
├── base_asset: string
├── quote_asset: string | null
├── market: string
├── interval: string
├── source: string
├── source_url: string | null
├── status: string

        ├── timestamp_raw: string | null
        ├── value: number | string | boolean | null
        │
        └── values: object
            │
            └── <metric>: number | string | boolean | null
